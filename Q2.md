**1. Start the table with each metro's average listing price**

```r
Q2_data_set <- postCOVID_housing_without_QF %>%
  group_by(cbsa_code) %>%
  summarise(cbsa_title          = first(cbsa_title),
            census_region       = first(census_region),
            household_size_tier = first(household_size_tier),
            avg_price           = mean(median_listing_price, na.rm = TRUE))
```

**2. Add average price per square foot**

```r
price_sqft <- postCOVID_housing_without_QF %>%
  group_by(cbsa_code) %>%
  summarise(avg_price_sqft = mean(median_listing_price_per_square_foot, na.rm = TRUE))

Q2_data_set <- Q2_data_set %>%
  left_join(price_sqft, by = "cbsa_code")
```

**3. Drop metros missing an average price**

```r
Q2_data_set <- Q2_data_set %>%
  filter(!is.na(avg_price))
```

**4. Add the log of average price**

```r
Q2_data_set <- Q2_data_set %>%
  mutate(log_avg_price = log(avg_price))
```

**5. Add the percent difference from the national average**

```r
national_avg <- mean(Q2_data_set$avg_price)

Q2_data_set <- Q2_data_set %>%
  mutate(pct_diff_national = (avg_price - national_avg) / national_avg * 100)

national_avg
```
